function setFade(root,k){
  root.userData.fadeK=k;
  root.traverse(o=>{
    if(o.isLight){ if(o.userData.baseI===undefined)o.userData.baseI=o.intensity;
      o.intensity=o.userData.baseI*k; return; }
    const m=o.material; if(!m)return;
    if(m.isShaderMaterial){ if(m.uniforms&&m.uniforms.uFade)m.uniforms.uFade.value=k; return; }
    if(m.userData.baseOpacity===undefined){ m.userData.baseOpacity=m.opacity; m.transparent=true; }
    m.opacity=m.userData.baseOpacity*k;
  });
}
function disposeGroup(root){
  root.traverse(o=>{
    if(o.geometry)o.geometry.dispose();
    if(o.material){
      const ms=Array.isArray(o.material)?o.material:[o.material];
      ms.forEach(m=>{ if(m.map)m.map.dispose(); m.dispose(); });
    }
  });
}

const PENTA=[261.6,293.7,329.6,392,440,523.3];
function initAudio(){
  if(groupAudio.ctx)return;
  try{
    const ctx=new (window.AudioContext||window.webkitAudioContext)();
    groupAudio.ctx=ctx;
    groupAudio.master=ctx.createGain(); groupAudio.master.gain.value=groupAudio.muted?0:1;
    groupAudio.master.connect(ctx.destination);
    const len=ctx.sampleRate*2, buf=ctx.createBuffer(1,len,ctx.sampleRate);
    const d=buf.getChannelData(0); let lastV=0;
    for(let i=0;i<len;i++){ const w=Math.random()*2-1; lastV=(lastV+0.02*w)/1.02; d[i]=lastV*3.5; }
    const src=ctx.createBufferSource(); src.buffer=buf; src.loop=true;
    const lp=ctx.createBiquadFilter(); lp.type='lowpass'; lp.frequency.value=300;
    groupAudio.river=ctx.createGain(); groupAudio.river.gain.value=0;
    src.connect(lp).connect(groupAudio.river).connect(groupAudio.master);
    src.start();
  }catch(e){}
}
function setAmbience(v){
  if(!groupAudio.river)return;
  const now=groupAudio.ctx.currentTime;
  groupAudio.river.gain.cancelScheduledValues(now);
  groupAudio.river.gain.linearRampToValueAtTime(v,now+1.6);
}
const _pluckCache={};
function pluck(noteIdx,delay=0,vol=0.18){
  if(!groupAudio.ctx)return;
  const ctx=groupAudio.ctx;
  const freq=PENTA[noteIdx%PENTA.length];
  const key=Math.round(freq);
  if(!_pluckCache[key]){
    const sr=ctx.sampleRate, N=Math.max(2,Math.round(sr/freq));
    const b=ctx.createBuffer(1,sr*2,sr), o=b.getChannelData(0);
    for(let i=0;i<N;i++)o[i]=Math.random()*2-1;
    for(let i=N;i<o.length;i++)o[i]=(o[i-N]+o[i-N-1])/2*0.996;
    _pluckCache[key]=b;
  }
  const s=ctx.createBufferSource(); s.buffer=_pluckCache[key];
  const g=ctx.createGain(); g.gain.value=vol;
  s.connect(g).connect(groupAudio.master);
  s.start(ctx.currentTime+delay);
}
function bell(){
  if(!groupAudio.ctx)return;
  const ctx=groupAudio.ctx, now=ctx.currentTime;
  [1244,1864].forEach((f,i)=>{
    const o=ctx.createOscillator(), g=ctx.createGain();
    o.frequency.value=f*(1+i*0.002); o.type='sine';
    g.gain.setValueAtTime(0.12/(i+1),now);
    g.gain.exponentialRampToValueAtTime(0.0001,now+0.9);
    o.connect(g).connect(groupAudio.master); o.start(now); o.stop(now+1);
  });
}

let zhVoice=null,curAudio=null,speakSeq=0,queueTimer=null;
function pickVoice(){
  if(!('speechSynthesis' in window))return;
  const vs=speechSynthesis.getVoices();
  const zh=vs.filter(v=>(v.lang||'').toLowerCase().startsWith('zh'));
  zhVoice=zh.find(v=>/Yunjian/i.test(v.name))||zh.find(v=>/Yunxi/i.test(v.name))
    ||zh.find(v=>/Xiaoxiao/i.test(v.name))||zh.find(v=>/Yunyang/i.test(v.name))
    ||zh.find(v=>/Natural|Online|Neural/i.test(v.name))||zh[0]||null;
}
if('speechSynthesis' in window){ speechSynthesis.onvoiceschanged=pickVoice; pickVoice(); }
const isNeural=v=>!!v&&/Yunjian|Yunxi|Xiaoxiao|Yunyang|Natural|Online|Neural/i.test(v.name||'');
function stopSpeak(){
  speakSeq++;
  if(curAudio){ try{curAudio.pause();}catch(e){} curAudio=null; }
  clearTimeout(queueTimer);
  if('speechSynthesis' in window)speechSynthesis.cancel();
}
function splitSentences(t){
  const out=[];let cur='';
  for(const ch of t){cur+=ch;if('。！？；'.includes(ch)){out.push(cur);cur='';}}
  if(cur)out.push(cur);return out.length?out:[t];
}
function speakLive(text,onend,seq,onFail){
  const parts=splitSentences(text);let i=0,failed=false;
  const fail=()=>{if(failed||seq!==speakSeq)return;failed=true;onFail&&onFail();};
  const next=()=>{
    if(seq!==speakSeq||failed)return;
    if(i>=parts.length){onend&&onend();return;}
    const u=new SpeechSynthesisUtterance(parts[i++]);
    u.lang='zh-CN';u.rate=0.78;u.pitch=1;u.volume=1;if(zhVoice)u.voice=zhVoice;
    u.onend=()=>{if(seq!==speakSeq||failed)return;if(i<parts.length)queueTimer=setTimeout(next,420);else next();};
    u.onerror=()=>{if(i===1)fail();else{if(i<parts.length)queueTimer=setTimeout(next,300);else onend&&onend();}};
    speechSynthesis.speak(u);
  };
  next();
}
function speak(text,onend){
  if(!('speechSynthesis' in window)){ tip('当前浏览器不支持朗读'); return; }
  speechSynthesis.cancel();
  const u=new SpeechSynthesisUtterance(text);
  u.lang='zh-CN'; u.rate=0.82; if(zhVoice)u.voice=zhVoice;
  if(onend)u.addEventListener('end',onend);
  speechSynthesis.speak(u);
}
function playMp3(file,fallbackText,onend){
  try{
    const a=new Audio('audio/'+file);
    curAudio=a; let done=false;
    const fallback=()=>{ if(done)return; done=true; curAudio=null; speak(fallbackText,onend); };
    a.addEventListener('error',fallback);
    a.addEventListener('ended',()=>{ if(!done){done=true;curAudio=null;onend&&onend();} });
    const p=a.play();
    if(p&&p.catch)p.catch(fallback);
  }catch(e){ speak(fallbackText,onend); }
}
function aiSpeak(file,fallbackText,onend){
  stopSpeak();
  const seq=speakSeq;
  if('speechSynthesis' in window&&zhVoice&&isNeural(zhVoice)){
    speakLive(fallbackText,onend,seq,()=>{ if(seq!==speakSeq)return; playMp3(file,fallbackText,onend); });
    return;
  }
  playMp3(file,fallbackText,onend);
}

/* ---------------- 主循环 ---------------- */
function animate(){
  requestAnimationFrame(animate);
  const now=performance.now()/1000;
  let dt=clock.last?now-clock.last:0.016;
  clock.last=now; dt=Math.min(dt,0.05); clock.t+=dt;
  const t=clock.t;
  if(state==='transition'&&trans){
    trans.t+=dt;
    const k=ease(clamp(trans.t/trans.dur,0,1));
    camera.position.lerpVectors(trans.fromP,trans.toP,k);
    curLook.lerpVectors(trans.fromL,trans.toL,k);
    setFade(trans.stage.group,k);
    if(trans.old)setFade(trans.old.group,1-k);
    mixSky(k);
    if(trans.stage.update)trans.stage.update(trans.t,dt);
    if(k>=1){
      if(trans.old){ scene.remove(trans.old.group); disposeGroup(trans.old.group); }
      curStageObj=trans.stage; curDef=STAGES[curIdx];
      state='stage'; stageT=0; autoT=0;
    }
  }
  else if(state==='stage'&&curStageObj){
    stageT+=dt; autoT+=dt;
    if(curStageObj.update)curStageObj.update(stageT,dt);
    const p=Math.min(1,stageT/30);
    camera.position.set(lerp(curDef.cam.f[0],curDef.cam.t[0],p),
      lerp(curDef.cam.f[1],curDef.cam.t[1],p),lerp(curDef.cam.f[2],curDef.cam.t[2],p));
    curLook.set(lerp(curDef.cam.lf[0],curDef.cam.lt[0],p),
      lerp(curDef.cam.lf[1],curDef.cam.lt[1],p),lerp(curDef.cam.lf[2],curDef.cam.lt[2],p));
    if(autoMode&&curIdx>0){
      if(autoT>curDef.dwell){ if(curIdx===8)showEnding(); else goto(curIdx+1); }
    }
  }
  if(curDef){
    const fwd=new THREE.Vector3().subVectors(curLook,camera.position).normalize();
    const right=new THREE.Vector3().crossVectors(fwd,new THREE.Vector3(0,1,0)).normalize();
    const up=new THREE.Vector3().crossVectors(right,fwd).normalize();
    camera.position.addScaledVector(right,mouse.x*2.4).addScaledVector(up,mouse.y*1.4);
    camera.position.y+=Math.sin(t*0.5)*0.16;
    camera.lookAt(curLook);
    if(curDef.roll&&state==='stage')camera.rotateZ(Math.sin(t*0.2)*curDef.roll);
  }
  if(bgRange)bgRange.update(t,mouse.x);
  applySky();
  renderer.render(scene,camera);
}
