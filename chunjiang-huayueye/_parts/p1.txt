/* 循文入境 · 春江花月夜 —— Three.js 沉浸式诗词课件
   八境随诗句推进，含注音/释义/注释/朗读/小测
   美术方案：青绿春晓（青绿山水 + 空明月色 + 花瓣/霜霰母题），标志性瞬间＝「江天一色 · 孤月轮」 */
'use strict';
window.__inited = false;

/* ---------------- 工具 ---------------- */
const $ = s => document.querySelector(s);
const clamp = (x,a,b)=>Math.max(a,Math.min(b,x));
const lerp = (a,b,t)=>a+(b-a)*t;
const sstep = (a,b,x)=>{ const t=clamp((x-a)/(b-a),0,1); return t*t*(3-2*t); };
const ease = t => t<0.5 ? 4*t*t*t : 1-Math.pow(-2*t+2,3)/2;
const v3 = a => new THREE.Vector3(a[0],a[1],a[2]);
const C = h => new THREE.Color(h);
const rnd = (a,b)=>a+Math.random()*(b-a);
const WHITE_RGB={r:1,g:1,b:1};   // 纯数据（顶层禁止求值 THREE）
const br = (t,period,phase)=>{ const p=(t/(period||11)+(phase||0))*Math.PI*2; return 0.5+0.5*Math.sin(p); };  // 潮汐/月华的"呼吸"

/* ---------------- 着色器 ---------------- */
const DOME_VERT=`varying vec3 vP; void main(){ vP=position; gl_Position=projectionMatrix*modelViewMatrix*vec4(position,1.0); }`;
const DOME_FRAG=`uniform vec3 uTop; uniform vec3 uHor; uniform vec3 uBot; varying vec3 vP;
void main(){ float h=normalize(vP).y;
  vec3 c=mix(uHor,uTop,smoothstep(0.02,0.5,h));
  c=mix(uBot,c,smoothstep(-0.3,0.02,h));
  gl_FragColor=vec4(c,1.0); }`;

const WATER_VERT=`
uniform float uTime; uniform float uAmp; uniform float uFreq; uniform float uSpeed; uniform vec2 uFlow;
varying vec3 vPos; varying vec3 vN;
float H(vec2 p){
  float t=uTime*uSpeed; float h=0.0;
  h+=sin(dot(p,vec2(0.9,0.35))*uFreq+t*1.10)*0.55;
  h+=sin(dot(p,vec2(-0.6,0.8))*uFreq*1.9+t*1.65)*0.30;
  h+=sin(dot(p,vec2(0.2,-1.0))*uFreq*3.1+t*2.30)*0.15;
  vec2 q=p+uFlow*uTime;
  h+=sin(dot(q,vec2(1.0,0.15))*uFreq*0.7+t*1.30)*0.45;
  return h;
}
void main(){
  vec3 p=position; float e=0.9;
  float h=H(p.xz); p.y+=h*uAmp;
  float hx=H(p.xz+vec2(e,0.0)); float hz=H(p.xz+vec2(0.0,e));
  vN=normalize(vec3((h-hx)*uAmp, e, (h-hz)*uAmp));
  vec4 wp=modelMatrix*vec4(p,1.0); vPos=wp.xyz;
  gl_Position=projectionMatrix*viewMatrix*wp;
}`;
const WATER_FRAG=`
uniform vec3 uDeep; uniform vec3 uShallow; uniform vec3 uSkyC; uniform vec3 uMoonDir; uniform vec3 uMoonColor;
uniform float uSpec; uniform float uFade; uniform vec3 uFogColor; uniform float uFogDensity;
varying vec3 vPos; varying vec3 vN;
void main(){
  vec3 N=normalize(vN);
  vec3 V=normalize(cameraPosition-vPos);
  float fres=pow(1.0-max(dot(N,V),0.0),2.0);
  vec3 col=mix(uDeep,uShallow,fres*0.75);
  col=mix(col,uSkyC,fres*0.55);
  vec3 R=reflect(-normalize(uMoonDir),N);
  float rv=max(dot(R,V),0.0);
  col+=uMoonColor*(pow(rv,90.0)*uSpec+pow(rv,700.0)*1.8);
  float d=length(cameraPosition-vPos);
  float f=1.0-exp(-uFogDensity*uFogDensity*d*d);
  col=mix(col,uFogColor,clamp(f,0.0,1.0));
  gl_FragColor=vec4(col,uFade);
}`;

/* 月华光路 uMoonPath：水面上一道随波晃动的月光（滟滟随波 / 月华流照 / 落月摇情 共用母题） */
const MPATH_VERT=`varying vec2 vUv; void main(){ vUv=uv; gl_Position=projectionMatrix*modelViewMatrix*vec4(position,1.0); }`;
const MPATH_FRAG=`
uniform float uTime; uniform float uFade; uniform float uOp; uniform float uWob; uniform float uNarrow;
uniform vec3 uColor; varying vec2 vUv;
void main(){
  float y=clamp(vUv.y,0.0,1.0);
  float wob=sin(y*34.0-uTime*1.7)*0.5+0.5;
  float wob2=sin(y*13.0-uTime*0.9+2.1)*0.5+0.5;
  float cx=0.5+(wob-0.5)*0.085*uWob;
  float dx=abs(vUv.x-cx)*2.0;
  float narrow=mix(1.0,0.42,uNarrow*y);
  float edge=1.0-smoothstep(0.20,narrow,dx);
  float band=edge*(0.52+0.48*wob)*(0.72+0.28*wob2);
  float far=1.0-smoothstep(0.62,1.0,y);
  float near=smoothstep(0.0,0.09,y);
  float a=band*far*near*uOp*uFade;
  if(a<0.004) discard;
  gl_FragColor=vec4(uColor*(0.78+0.42*wob),a);
}`;

/* 涟漪 uRipple：点击江面／鱼龙潜跃——同心圆一圈圈漫开 */
const RIP_VERT=`varying vec2 vUv; void main(){ vUv=uv; gl_Position=projectionMatrix*modelViewMatrix*vec4(position,1.0); }`;
const RIP_FRAG=`
uniform float uTime; uniform float uFade; uniform float uT0; uniform float uMaxR; uniform vec3 uColor; uniform float uOp;
varying vec2 vUv;
void main(){
  vec2 p=(vUv-0.5)*2.0*uMaxR;
  float r=length(p);
  float age=max(uTime-uT0,0.0);
  float f=1.0-clamp(age/2.8,0.0,1.0);
  float a=0.0;
  float w=max(uMaxR*0.035,0.9);
  for(int i=0;i<3;i++){
    float fi=float(i);
    float R=age*uMaxR*(0.34-fi*0.085);
    a+=smoothstep(w,0.0,abs(r-R))*(1.0-fi*0.28)*step(0.001,R);
  }
  a*=f*f*uFade*uOp;
  if(a<0.004) discard;
  gl_FragColor=vec4(uColor,a);
