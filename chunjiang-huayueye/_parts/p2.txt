function makeStars(n,size,op){
  const g=new THREE.BufferGeometry(); const P=new Float32Array(n*3); const Cc=new Float32Array(n*3);
  for(let i=0;i<n;i++){
    const a=Math.random()*Math.PI*2, e=Math.asin(rnd(-0.04,1)), r=410;
    P[i*3]=Math.cos(e)*Math.sin(a)*r; P[i*3+1]=Math.sin(e)*r; P[i*3+2]=Math.cos(e)*Math.cos(a)*r;
    const w=rnd(0.75,1); const tint=Math.random();
    Cc[i*3]=w*(0.86+tint*0.14); Cc[i*3+1]=w; Cc[i*3+2]=w*(0.90+tint*0.10);
  }
  g.setAttribute('position',new THREE.BufferAttribute(P,3));
  g.setAttribute('color',new THREE.BufferAttribute(Cc,3));
  const m=new THREE.PointsMaterial({size,sizeAttenuation:false,map:circleTex(),vertexColors:true,
    transparent:true,opacity:op,depthWrite:false,fog:false,blending:THREE.AdditiveBlending});
  m.userData.baseOp=op;
  const p=new THREE.Points(g,m); p.renderOrder=-9; return p;
}

function mixSky(k){
  if(!skyFrom||!skyTo)return;
  SKYcur.top.lerpColors(skyFrom.top,skyTo.top,k);
  SKYcur.hor.lerpColors(skyFrom.hor,skyTo.hor,k);
  SKYcur.bot.lerpColors(skyFrom.bot,skyTo.bot,k);
  SKYcur.fog.lerpColors(skyFrom.fog,skyTo.fog,k);
  SKYcur.dirC.lerpColors(skyFrom.dirC,skyTo.dirC,k);
  SKYcur.ambC.lerpColors(skyFrom.ambC,skyTo.ambC,k);
  SKYcur.moon.lerpVectors(skyFrom.moon,skyTo.moon,k);
  SKYcur.dirP.lerpVectors(skyFrom.dirP,skyTo.dirP,k);
  SKYcur.fd=lerp(skyFrom.fd,skyTo.fd,k); SKYcur.star=lerp(skyFrom.star,skyTo.star,k);
  SKYcur.ms=lerp(skyFrom.ms,skyTo.ms,k); SKYcur.dirI=lerp(skyFrom.dirI,skyTo.dirI,k);
  SKYcur.mph=lerp(skyFrom.mph,skyTo.mph,k); SKYcur.mhaze=lerp(skyFrom.mhaze,skyTo.mhaze,k);
  SKYcur.ambI=lerp(skyFrom.ambI,skyTo.ambI,k);
  applySky();
}

  renderer.setClearColor(SKYcur.fog);
  starA.material.opacity=SKYcur.star*(0.8+0.2*Math.sin(clock.t*0.7));
  starB.material.opacity=SKYcur.star*(0.55+0.25*Math.sin(clock.t*0.53+2));
  moonGroup.position.copy(SKYcur.moon);
  moonGroup.scale.setScalar(SKYcur.ms);
  moonGlow.material.color.copy(SKYcur.dirC).lerp(WHITE_RGB,0.68);
  moonCtl.set({glowColor:moonGlow.material.color,phase:SKYcur.mph,haze:SKYcur.mhaze});
  moonCtl.update(clock.t);
  dirLight.color.copy(SKYcur.dirC); dirLight.intensity=SKYcur.dirI; dirLight.position.copy(SKYcur.dirP);

  for(const u of fogShaders){ if(u.uFogColor&&u.uFogDensity){ u.uFogColor.value.copy(scene.fog.color); u.uFogDensity.value=scene.fog.density; } }

/* ---------------- 造型工坊（全部惰性：THREE 对象只在函数内创建） ----------------
   手法：LatheGeometry 立器型/袍身、mergeGeos 顶点色合批（一个人物 3 draw call / 一件器物 1-2）、
   rimHook 边缘光（夜景里靠轮廓+边缘光把人从暗底里"跳"出来）、一切每帧写 opacity/intensity 都乘 fadeK。 */
function seedRnd(seed){ let s=(seed>>>0)||1; return function(){ s=(s*1664525+1013904223)>>>0; return s/4294967296; }; }
function shadeColor(c,k){
  const r=Math.min(255,Math.round(((c>>16)&255)*k)),g=Math.min(255,Math.round(((c>>8)&255)*k)),b=Math.min(255,Math.round((c&255)*k));
  return (r<<16)|(g<<8)|b;
}
/* 合批：若干几何体（各带一个顶点色）并成一个带 color 属性的 BufferGeometry —— 省 draw call 的关键 */
function mergeGeos(list){
  const gs=list.map(function(it){
    let g=it.geo; if(g.index)g=g.toNonIndexed();
    const n=g.attributes.position.count;
    if(!g.attributes.normal)g.setAttribute('normal',new THREE.BufferAttribute(new Float32Array(n*3),3));
    if(!g.attributes.uv)g.setAttribute('uv',new THREE.BufferAttribute(new Float32Array(n*2),2));
    const c=it.color,r=((c>>16)&255)/255,gg=((c>>8)&255)/255,b=(c&255)/255;
    const col=new Float32Array(n*3);
    for(let i=0;i<n;i++){col[i*3]=r;col[i*3+1]=gg;col[i*3+2]=b;}
    g.setAttribute('color',new THREE.BufferAttribute(col,3));
    return g;
  });
  let n=0; gs.forEach(function(g){n+=g.attributes.position.count;});
  const pos=new Float32Array(n*3),nor=new Float32Array(n*3),uv=new Float32Array(n*2),col=new Float32Array(n*3);
  let o=0;
  gs.forEach(function(g){
    pos.set(g.attributes.position.array,o*3);
    nor.set(g.attributes.normal.array,o*3);
    uv.set(g.attributes.uv.array,o*2);
    col.set(g.attributes.color.array,o*3);
    o+=g.attributes.position.count;
  });
  const out=new THREE.BufferGeometry();
  out.setAttribute('position',new THREE.BufferAttribute(pos,3));
  out.setAttribute('normal',new THREE.BufferAttribute(nor,3));
  out.setAttribute('uv',new THREE.BufferAttribute(uv,2));
  out.setAttribute('color',new THREE.BufferAttribute(col,3));
  return out;
}

/* 边缘光：给 Phong 材质注入菲涅尔 rim 项（同一份 onBeforeCompile 源码 → 共用一个着色程序） */
function rimOnBeforeCompile(sh){
  const r=(this&&this.userData&&this.userData.rim)||{c:0x9ec9b8,p:2.4,i:0.45};
  sh.uniforms.uRimC={value:new THREE.Color(r.c)};
  sh.uniforms.uRimP={value:r.p};
  sh.uniforms.uRimI={value:r.i};
  sh.fragmentShader='uniform vec3 uRimC;uniform float uRimP;uniform float uRimI;\n'+sh.fragmentShader;
  sh.fragmentShader=sh.fragmentShader.replace('#include <dithering_fragment>',
    '#include <dithering_fragment>\n{ vec3 _V=normalize(vViewPosition); vec3 _N=normalize(vNormal);'+
    ' float _rm=pow(1.0-clamp(dot(_N,_V),0.0,1.0),uRimP); gl_FragColor.rgb+=uRimC*_rm*uRimI; }');
}
function rimHook(mat,opt){
  const o=opt||{};
  mat.userData.rim={c:o.c===undefined?0x9ec9b8:o.c,p:o.p===undefined?2.4:o.p,i:o.i===undefined?0.45:o.i};
  mat.onBeforeCompile=rimOnBeforeCompile;
  return mat;
}
/* 两点之间的一段"肢体"（r128 没有 CapsuleGeometry：圆柱收细 + 球关节） */
function limbGeo(a,b,r0,r1,seg){
  const A=new THREE.Vector3(a[0],a[1],a[2]),B=new THREE.Vector3(b[0],b[1],b[2]);
  const d=new THREE.Vector3().subVectors(B,A), len=Math.max(d.length(),0.0001);
  const g=new THREE.CylinderGeometry(r1,r0,len,seg||8,1,false);
  g.translate(0,len/2,0);
  const q=new THREE.Quaternion().setFromUnitVectors(new THREE.Vector3(0,1,0),d.clone().normalize());
  const m4=new THREE.Matrix4().makeRotationFromQuaternion(q); m4.setPosition(A);
  g.applyMatrix4(m4);
  return g;
}
