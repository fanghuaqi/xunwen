const VES_MAT={
  '金':{c:0xc9a24a,s:0xfff2c8,sh:80,e:0x1c1408,rimC:0xffe2a0,rim:0.50},
  '陶':{c:0x4b3d31,s:0x9a8060,sh:26,e:0x070605,rimC:0xd8c49a,rim:0.34},

function vesMat(kind){
  const d=VES_MAT[kind]||VES_MAT['陶'];
  const m=rimHook(new THREE.MeshPhongMaterial({color:0xffffff,vertexColors:true,shininess:d.sh,
    specular:d.s,emissive:d.e}),{c:d.rimC,i:d.rim,p:3.0});
  m.side=THREE.DoubleSide;
  if(d.op!==undefined){ m.transparent=true; m.opacity=d.op; }
  return m;
}
function makeVessel(o){
  o=o||{};
  const type=VES_PTS[o.type]?o.type:'杯';
  const kind=VES_MAT[o.mat]?o.mat:'陶';
  const d=VES_MAT[kind];
  const sc=o.scale===undefined?1:o.scale;
  const pts=VES_PTS[type], base=o.color===undefined?d.c:o.color;
  const g=new THREE.Group(), B=new GeoBag();
  B.put(new THREE.LatheGeometry(pts.map(p=>new THREE.Vector2(p[0],p[1])),o.seg||24),base);
  const rimR=pts[pts.length-1][0], rimY=pts[pts.length-1][1];
  const ring=new THREE.TorusGeometry(rimR,Math.max(rimR*0.05,0.013),6,24);
  ring.rotateX(Math.PI/2); ring.translate(0,rimY,0); B.put(ring,shadeColor(base,1.5));
  const nLeg=(type==='樽'||type==='爵')?3:0;
  for(let i=0;i<nLeg;i++){
    const a=i/nLeg*Math.PI*2+Math.PI/2;
    const lg=new THREE.CylinderGeometry(0.048*sc,0.075*sc,pts[0][1],5);
    lg.translate(Math.sin(a)*rimR*0.60,pts[0][1]*0.5,Math.cos(a)*rimR*0.60);
    B.put(lg,shadeColor(base,0.78));
  }
  if(type==='壶'){
    const hd=new THREE.TorusGeometry(0.19,0.033,6,14,Math.PI*1.15);
    hd.rotateY(Math.PI/2); hd.translate(-0.30,1.02,0); B.put(hd,shadeColor(base,0.85));
    const sp=new THREE.CylinderGeometry(0.05,0.085,0.40,7);
    sp.rotateZ(-1.02); sp.translate(0.29,1.20,0); B.put(sp,base);
    const lid=new THREE.ConeGeometry(0.24,0.16,10); lid.translate(0,1.40,0); B.put(lid,shadeColor(base,1.12));
    const kn=new THREE.SphereGeometry(0.055,8,6); kn.translate(0,1.50,0); B.put(kn,shadeColor(base,1.3));
  }
  if(type==='坛'){
    const cl=new THREE.ConeGeometry(0.36,0.30,12); cl.translate(0,1.68,0); B.put(cl,0x8c2f22);
    const kt=new THREE.SphereGeometry(0.09,8,6); kt.translate(0,1.84,0); B.put(kt,0x6f2419);
  }
  const mesh=B.mesh(vesMat(kind)); g.add(mesh);
  if(o.shadow!==false){
    let rr=0.3; pts.forEach(p=>{ if(p[0]>rr)rr=p[0]; });
    const sh=new THREE.Mesh(new THREE.CircleGeometry(rr*1.45,18),
      new THREE.MeshBasicMaterial({map:glowTex(),color:0x000000,transparent:true,opacity:0.55,depthWrite:false}));
    sh.rotation.x=-Math.PI/2; sh.position.y=0.012; sh.renderOrder=0; g.add(sh);
  }
  let wine=null;
  if(o.liquid){
    wine=new THREE.Mesh(new THREE.CircleGeometry(rimR*0.86,20),
      new THREE.MeshPhongMaterial({color:0x5a6a3a,specular:0xd8f0dc,shininess:120,
        emissive:0x0c1408,transparent:true,opacity:0.92}));
    wine.rotation.x=-Math.PI/2; wine.position.y=rimY-0.055; wine.renderOrder=2; g.add(wine);
  }
  g.scale.setScalar(sc);
  const ph=Math.random()*6.283;
  return {g,mesh,wine,type,rimY:rimY*sc,update(t,fk){
    if(wine){ wine.scale.setScalar(1+0.022*Math.sin(t*2.4+ph)); }
  },setLiquid(y){ if(wine)wine.position.y=y; }};
}

function makeCrowd(o){
  o=o||{};
  const n=o.n===undefined?10:o.n, R=seedRnd(o.seed===undefined?11:o.seed);
  const rect=o.rect||[-30,6,60,14];
  const mesh=new THREE.InstancedMesh(crowdGeo(),
    rimHook(new THREE.MeshPhongMaterial({color:o.color===undefined?0x141d1a:o.color,vertexColors:true,shininess:10,
      specular:0x243430,emissive:0x030706}),{c:o.rimC===undefined?0x9ec9b8:o.rimC,i:o.rim===undefined?0.22:o.rim,p:2.6}),n);
  const dm=new THREE.Object3D(), base=[];
  for(let i=0;i<n;i++){
    const x=rect[0]+R()*rect[2], z=rect[1]+R()*rect[3], s=(o.sMin||0.85)+R()*((o.sMax||1.15)-(o.sMin||0.85));
    base.push({x,y:0,z,s,ry:R()*6.283,ph:R()*6.283});
  }
  mesh.frustumCulled=false;
  const put=function(t){
    for(let i=0;i<n;i++){
      const b=base[i];
      dm.position.set(b.x,b.y+0.02*Math.sin(t*1.1+b.ph),b.z);
      dm.position.y+=(o.y||0);
      dm.rotation.set(0,b.ry+0.02*Math.sin(t*0.3+b.ph),0);
      dm.scale.setScalar(b.s*(o.walk?1:1));
      if(o.walk)dm.position.x+=Math.sin(t*0.10+b.ph)*6.0;
      dm.updateMatrix(); mesh.setMatrixAt(i,dm.matrix);
    }
    mesh.instanceMatrix.needsUpdate=true;
  };
  put(0);
  return {mesh,update:put};
}

/* ================= 火 makeFlame(o) ================= */
function makeFlame(o){
  o=o||{};
  const h=o.h===undefined?3:o.h, w=o.w===undefined?h*0.46:o.w;
  const core=o.core===undefined?0xfff0b0:o.core, outer=o.outer===undefined?0xff5a14:o.outer;
  const mat=new THREE.ShaderMaterial({transparent:true,depthWrite:false,side:THREE.DoubleSide,
    blending:THREE.AdditiveBlending,
    uniforms:{uTime:{value:0},uFade:{value:0},uCore:{value:C(core)},uOuter:{value:C(outer)},
      uSeed:{value:o.seed===undefined?Math.random()*10:o.seed},uWide:{value:o.wide===undefined?0.30:o.wide}},
    vertexShader:FLAME_VERT,fragmentShader:FLAME_FRAG});
  const planes=o.planes===undefined?3:o.planes, B=new GeoBag();
  for(let i=0;i<planes;i++){
    const p=new THREE.PlaneGeometry(w,h,1,1);
    p.translate(0,h/2,0); p.rotateY(i*Math.PI/planes);
    B.put(p,0xffffff);
  }
  const g=new THREE.Group();
  const mesh=new THREE.Mesh(mergeGeos(B.list),mat);
  mesh.renderOrder=4; mesh.frustumCulled=false; g.add(mesh);
  const emb=makeGlow({n:o.embers===undefined?46:o.embers,box:[w*0.45,h*0.6,w*0.45],pos:[0,h*0.22,0],
    color:core,size:o.esize||5,speed:o.espeed||0.9,rise:1,maxA:0.85});
  emb.points.renderOrder=4; g.add(emb.points);
  const burst=o.spark===false?null:makeBurst({n:o.sparkN||22,color:core,pos:[0,h*0.34,0]});
  if(burst)g.add(burst.points);
  let light=null, base=1;
  if(o.light){
    light=new THREE.PointLight(o.lightC===undefined?0xff8c42:o.lightC,o.light,o.light===undefined?60:(o.lightD||60));
    light.position.set(0,h*0.55,0); g.add(light); base=o.light;
  }
  let next=1.2+Math.random()*2.2;
  const s1=Math.random()*9, s2=Math.random()*9;
  g.update=function(t,fk){
    const k=fk===undefined?1:fk;
    mat.uniforms.uFade.value=k;
    mat.uniforms.uTime.value=t;
    emb.update(t);
    if(burst){ burst.update(t); if(t>next){ burst.fire(); next=t+1.8+Math.random()*3.6; } }
    if(light)light.intensity=base*(0.80+0.20*Math.sin(t*11.3+s1)+0.12*Math.sin(t*27.7+s2))*k;
  };
  g.userData.update=g.update;
  return {g,mat,update:g.update};
}

/* ================= 远山 makeRange(o) ================= */
function makeRange(o){
  o=o||{};
  const layers=o.layers===undefined?3:o.layers;
  const r0=o.r===undefined?300:o.r, arc=o.arc===undefined?Math.PI*2:o.arc;
  const a0=o.a0===undefined?0:o.a0, yBase=o.y===undefined?-16:o.y;
  const hBase=o.h===undefined?72:o.h, seg=o.seg===undefined?150:o.seg;
  const base=o.color===undefined?0x06120f:o.color, atmo=o.atmo===undefined?0x2c6055:o.atmo;
  const R=seedRnd(o.seed===undefined?7:o.seed);
  const g=new THREE.Group(), items=[];
  for(let L=0;L<layers;L++){
    const tL=layers<2?1:L/(layers-1);
    const rr=r0*(1.0-0.22*tL), hh=hBase*(0.60+0.58*tL);
    const peaks=[], np=o.peaks===undefined?6:o.peaks;
    for(let i=0;i<np;i++) peaks.push({a:a0+arc*(i+0.12+R()*0.76)/np, h:hh*(0.40+0.60*R()),
      w:arc*(0.055+R()*0.105), p:0.72+R()*0.85});
    for(let i=0;i<16;i++) peaks.push({a:a0+arc*R(), h:hh*(0.08+0.22*R()), w:arc*(0.014+R()*0.032), p:1.3});
    const f=5+R()*7, ph=R()*6.283, jit=hh*0.026;
    const pos=[],uv=[],idx=[];
    for(let i=0;i<=seg;i++){
      const ang=a0+arc*i/seg;
      let h=0;
      for(let k=0;k<peaks.length;k++){
        const pk=peaks[k];
        const da=Math.abs(((ang-pk.a+Math.PI*3)%(Math.PI*2))-Math.PI), x=da/pk.w;
        if(x<1)h+=pk.h*Math.pow(1-x,pk.p);
      }
      h*=1.0+0.14*Math.sin(ang*f+ph);
      h+=jit*(Math.sin(ang*13.7+ph*3)+0.6*Math.sin(ang*29.3+ph*7));

      pos.push(sx,yBase,sz, sx,yBase+Math.max(h,1.5),sz);
      uv.push(i/seg,0, i/seg,1);
    }
    for(let i=0;i<seg;i++){ const a=i*2; idx.push(a,a+2,a+1, a+1,a+2,a+3); }
    const geo=new THREE.BufferGeometry();
    geo.setAttribute('position',new THREE.BufferAttribute(new Float32Array(pos),3));
    geo.setAttribute('uv',new THREE.BufferAttribute(new Float32Array(uv),2));
    geo.setIndex(idx); geo.computeVertexNormals();
    const mt=new THREE.ShaderMaterial({transparent:true,side:THREE.DoubleSide,
      uniforms:{uLow:{value:C(base).lerp(C(atmo),0.30*(1-tL))},
        uHigh:{value:C(base).lerp(C(atmo),0.62*(1-tL)+0.10)},
        uGlow:{value:C(o.glow===undefined?0xcfe8dc:o.glow)},
        uFogColor:{value:C(0x0e1d16)},uFogDensity:{value:0.005},
        uFogK:{value:o.fogK===undefined?0.62:o.fogK},
        uGlowK:{value:(o.glowK===undefined?0.07:o.glowK)*(0.30+0.70*tL)},
        uFade:{value:1}},
      vertexShader:RANGE_VERT,fragmentShader:RANGE_FRAG});
    fogShaders.push(mt.uniforms);
    const mesh=new THREE.Mesh(geo,mt);
    mesh.frustumCulled=false; mesh.renderOrder=o.order===undefined?-6:o.order;
    g.add(mesh); items.push({mesh,r:rr});
  }
  g.update=function(t,px){ const p=px||0;
    for(let i=0;i<items.length;i++) items[i].mesh.position.x=p*0.5*(i+1); };
  g.userData.update=g.update;
  return {g,update:g.update,items};
}

    uniforms:{uBase:{value:C(o.base===undefined?0xeaf6ee:o.base)},

  });
  group.add(disc);
  group.renderOrder=-8;
  const api={group,disc,glow:halo[0].s,mat,halo,
    set:function(p){
      if(p.color)mat.uniforms.uBase.value.copy(p.color);
      if(p.phase!==undefined)mat.uniforms.uPhase.value=p.phase;
      if(p.haze!==undefined)mat.uniforms.uHazeK.value=p.haze;

        halo[1].s.material.color.copy(p.glowColor); }

        for(let i=0;i<halo.length;i++)halo[i].op=halo[i].s.material.opacity; }

function rockGeo(r,detail,R){
  const g=new THREE.IcosahedronGeometry(r,detail===undefined?1:detail);
  const p=g.attributes.position, a=p.array;
  for(let i=0;i<a.length;i+=3){
    const x=a[i],y=a[i+1],z=a[i+2];
    let h=Math.sin(x*12.9898+y*78.233+z*37.719)*43758.5453; const f1=h-Math.floor(h);
    h=Math.sin(x*39.346+y*11.135+z*83.155)*24634.6345; const f2=h-Math.floor(h);
    const k=0.70+0.50*f1;
    a[i]=x*k; a[i+1]=y*(0.66+0.42*f2); a[i+2]=z*k;
  }
  p.needsUpdate=true; g.computeVertexNormals();
  return g;
