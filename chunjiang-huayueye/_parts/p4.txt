
/* ================= 花林/青枫 makeGrove(o) =================
   树干枝条合并 1 mesh + 花叶团合并 1 mesh（整片林子 2 draw call）。
   leaf:花/叶色；bloom:false 时少分叉（青枫的叶色更实）；crown:花团半径 */
function makeGrove(o){
  o=o||{};
  const n=o.n===undefined?10:o.n, R=seedRnd(o.seed===undefined?17:o.seed);
  const trunk=o.trunk===undefined?0x2a2418:o.trunk, leaf=o.leaf===undefined?0xefdde3:o.leaf;
  const spread=o.spread||[26,14], hh0=o.h===undefined?5.2:o.h, cr0=o.crown===undefined?1.5:o.crown;
  const TB=new GeoBag(), LB=new GeoBag();
  for(let i=0;i<n;i++){
    const x=(R()-0.5)*spread[0], z=(R()-0.5)*spread[1];
    const h=hh0*(0.72+0.6*R()), r0=(o.r===undefined?0.28:o.r)*(0.8+0.5*R());
    const tx=x+(R()-0.5)*h*0.16, tz=z+(R()-0.5)*h*0.12;
    TB.put(limbGeo([x,0,z],[x+(tx-x)*0.6,h*0.60,z+(tz-z)*0.6],r0,r0*0.66,6),shadeColor(trunk,0.85+0.3*R()));
    TB.put(limbGeo([x+(tx-x)*0.6,h*0.58,z+(tz-z)*0.6],[tx,h,tz],r0*0.64,r0*0.36,5),trunk);
    const nb=o.bloom===false?2:3;
    for(let b=0;b<nb;b++){
      const a=R()*6.283, by=h*(0.56+R()*0.34);
      const bx=x+(tx-x)*(by/h)+Math.cos(a)*h*0.32, bz=z+(tz-z)*(by/h)+Math.sin(a)*h*0.24;
      TB.put(limbGeo([x+(tx-x)*(by/h),by*0.94,z+(tz-z)*(by/h)],[bx,by,bz],r0*0.34,r0*0.15,4),trunk);
      const cr=cr0*(0.62+0.66*R());
      const cg=new THREE.IcosahedronGeometry(cr,1);
      cg.scale(1.22,0.70,1.12); cg.translate(bx,by+cr*0.22,bz);
      LB.put(cg,shadeColor(leaf,0.74+0.56*R()));
    }
    const cg2=new THREE.IcosahedronGeometry(cr0*1.22,1);
    cg2.scale(1.18,0.64,1.08); cg2.translate(tx,h,tz);
    LB.put(cg2,shadeColor(leaf,0.80+0.50*R()));
  }
  const g=new THREE.Group();
  g.add(TB.mesh(rimHook(new THREE.MeshPhongMaterial({color:0xffffff,vertexColors:true,shininess:8,
    specular:0x2c3a34,emissive:0x030605}),{c:o.rimC===undefined?0x9ec9b8:o.rimC,i:o.rim===undefined?0.30:o.rim,p:2.4})));
  g.add(LB.mesh(rimHook(new THREE.MeshPhongMaterial({color:0xffffff,vertexColors:true,shininess:18,
    specular:0x7a9a8c,emissive:o.leafEm===undefined?0x0e1a16:o.leafEm}),{c:o.rimC===undefined?0xd6f2e4:o.rimC,i:o.rim===undefined?0.44:o.rim,p:2.0})));
  g.update=function(t,fk){ const k=fk===undefined?1:fk; g.rotation.z=Math.sin(t*0.24+(o.seed||0))*0.006; };
  g.userData.update=g.update;
  return {g,update:g.update};
}

/* ================= 舟 makeBoat(o) =================
   船壳为 Lathe 剖面沿 z 压扁再削尖首尾；乌篷用半圆柱拱；可挂灯、可载人。 */
function makeBoat(o){
  o=o||{};
  const L=o.len===undefined?7.4:o.len, W=o.w===undefined?2.5:o.w, sc=o.scale===undefined?1:o.scale;
  const wood=o.wood===undefined?0x2b2418:o.wood;
  const B=new GeoBag();
  const hullPts=[[0,0],[0.42,0.06],[0.62,0.30],[0.60,0.58],[0.52,0.86],[0.30,1.00],[0.02,1.06]];
  const hull=new THREE.LatheGeometry(hullPts.map(p=>new THREE.Vector2(p[0]*W*0.5,p[1]*W*0.42)),22);
  hull.scale(1,1,0.34);                       // 压扁成梭形，长轴留 x
  const hp=hull.attributes.position;
  for(let i=0;i<hp.count;i++){                 // 首尾收尖
    const x=hp.getX(i), y=hp.getY(i), z=hp.getZ(i);
    const k=1.0-0.55*Math.min(1,Math.pow(Math.abs(x)/(W*0.5),2.2));
    hp.setZ(i,z*k); hp.setX(i,x);
    hp.setY(i,y+0.06*Math.min(1,Math.abs(x)/(W*0.4)));
  }
  hull.computeVertexNormals();
  B.put(hull,wood);
  const rim=new THREE.TorusGeometry(W*0.5,0.055,5,26); rim.rotateX(Math.PI/2); rim.scale(1,1,0.34);
  B.put(rim,shadeColor(wood,1.5));
  if(o.canopy!==false){
    const cp=new THREE.CylinderGeometry(W*0.36,W*0.36,L*0.46,14,1,true,0,Math.PI);
    cp.rotateZ(Math.PI/2); cp.scale(1,0.94,0.80); cp.translate(-L*0.04,0.26,0); B.put(cp,0x3a3a2a);
    const rib=new THREE.TorusGeometry(W*0.36,0.04,4,18,Math.PI);
    rib.rotateY(Math.PI/2); rib.scale(1,0.94,0.80); rib.translate(L*0.19,0.26,0); B.put(rib,shadeColor(wood,1.2));
    const rib2=rib.clone(); rib2.translate(-L*0.38,0,0); B.put(rib2,shadeColor(wood,1.2));
  }
  const bow=new THREE.BoxGeometry(0.16,1.5,0.16); bow.rotateX(0.22); bow.translate(L*0.44,0.75,0); B.put(bow,shadeColor(wood,1.1));
  const mesh=B.mesh(rimHook(new THREE.MeshPhongMaterial({color:0xffffff,vertexColors:true,shininess:14,
    specular:0x4a4a34,emissive:0x030504,side:THREE.DoubleSide}),{c:o.rimC===undefined?0x9ec9b8:o.rimC,i:o.rim===undefined?0.40:o.rim,p:2.4}));
  const g=new THREE.Group(); g.add(mesh);
  let lan=null;
  if(o.lantern){
    lan=makeLantern(o.lanternScale===undefined?0.30:o.lanternScale,{flame:o.flame!==false,glow:0.70});
    lan.position.set(-L*0.30,1.34,0); g.add(lan);
    const pole=new THREE.Mesh(new THREE.CylinderGeometry(0.05,0.06,1.5,6),
      new THREE.MeshPhongMaterial({color:shadeColor(wood,0.9),shininess:8}));
    pole.position.set(-L*0.30,1.02,0); g.add(pole);
  }
  const sh=new THREE.Mesh(new THREE.CircleGeometry(W*0.85,20),
    new THREE.MeshBasicMaterial({map:glowTex(),color:0x000000,transparent:true,opacity:0.42,depthWrite:false}));
  sh.rotation.x=-Math.PI/2; sh.position.y=0.02; sh.renderOrder=0; g.add(sh);
  g.scale.setScalar(sc);
  const ph=Math.random()*6.283;
  g.update=function(t,fk){ const k=fk===undefined?1:fk;
    g.position.y=(o.y||0)+Math.sin(t*0.62+ph)*0.10*sc;
    g.rotation.z=Math.sin(t*0.48+ph)*0.028;
    g.rotation.x=Math.sin(t*0.40+ph*1.4)*0.022;
    if(lan)lan.update(t,k);
  };
  g.userData.update=g.update;
  return {g,update:g.update,lantern:lan};
}

/* ================= 鸿雁 makeGeese(o) =================

function gooseGeo(){
  const B=new GeoBag();
  const w1=new THREE.PlaneGeometry(2.7,0.60); w1.translate(-1.55,0.34,0); B.put(w1,0xffffff);

function makeGeese(o){
  o=o||{};
  const n=o.n===undefined?9:o.n, sc=o.scale===undefined?1:o.scale;
  const mesh=new THREE.InstancedMesh(gooseGeo(),
    new THREE.MeshBasicMaterial({color:o.color===undefined?0x14231e:o.color,side:THREE.DoubleSide}),n);
  mesh.frustumCulled=false; mesh.renderOrder=2;
  const dm=new THREE.Object3D(), R=seedRnd(o.seed===undefined?29:o.seed);
  const base=[];
  for(let i=0;i<n;i++){
    const row=Math.ceil(i/2), side=(i%2)?1:-1;
    base.push({row,side,ph:R()*6.283});
  }
  const put=function(t){
    const cyc=(t*(o.speed===undefined?0.016:o.speed))%(o.span===undefined?1.25:o.span)-0.12;
    const cx=lerp(o.x0===undefined?-190:o.x0,o.x1===undefined?190:o.x1,cyc);
    const y0=o.y===undefined?64:o.y;
    for(let i=0;i<n;i++){
      const b=base[i];
      dm.position.set(cx-b.side*b.row*4.2*sc, y0-b.row*1.5*sc+Math.sin(t*0.55+b.ph)*1.5,
        (o.z===undefined?-150:o.z)+b.row*2.6*sc+Math.sin(t*0.4+b.ph)*1.2);
      dm.rotation.set(0,Math.PI*0.5,0);
      const f=Math.abs(Math.sin(t*(o.flap===undefined?2.3:o.flap)+b.ph));
      dm.scale.set(sc,sc*(0.28+0.90*f),sc);
      dm.updateMatrix(); mesh.setMatrixAt(i,dm.matrix);
    }
    mesh.instanceMatrix.needsUpdate=true;
  };
  put(0);
  return {mesh,update:put};
}
